import java.util.*;
class Solution {
    public boolean canJump(int[] nums) {
        int maxLen =0;
        int i = 0;
        int size = nums.length;
        while (i < size){
            if(i > maxLen)
                return false;
                maxLen = Math.max(maxLen, i+nums[i]);
                i++;
        }
         return true;
    }
}
